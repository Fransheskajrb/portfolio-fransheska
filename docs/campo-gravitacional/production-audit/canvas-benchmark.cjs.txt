const {chromium}=require('playwright');const ts=require('/workspace/portfolio-fransheska/node_modules/typescript');const fs=require('fs');
(async()=>{const b=await chromium.launch({executablePath:'/usr/bin/chromium',args:['--no-sandbox']});const results=[];for(const version of ['before','after'])for(const [name,width,height,cw,ch,top] of [['desktop',1440,900,835.2,760,100],['mobile',390,844,343.2,288.6,530]])for(let run=0;run<3;run++){
 const source=fs.readFileSync(version==='before'?'/tmp/campo-network-before.ts':'/workspace/portfolio-fransheska/src/lib/skillNetwork.ts','utf8');const js=ts.transpileModule(source,{compilerOptions:{module:ts.ModuleKind.CommonJS,target:ts.ScriptTarget.ES2020}}).outputText;

 await p.addScriptTag({content:js+';window.cleanup=exports.createSkillNetwork(document.querySelector("canvas"))'});await p.waitForTimeout(150);

 results.push({version,name,run,active,offscreenDraws:offscreen});fs.writeFileSync('/tmp/campo-canvas-comparison.json',JSON.stringify(results,null,2));console.log(JSON.stringify(results.at(-1)));await p.close();}
await b.close()})().catch(e=>{console.error(e);process.exit(1)});

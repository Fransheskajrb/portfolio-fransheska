const{chromium}=require('playwright');const fs=require('fs');
(async()=>{const b=await chromium.launch({executablePath:'/usr/bin/chromium',args:['--no-sandbox']});const rows=[];for(const[w,h]of[[1440,900],[768,1024],[390,844],[320,800]]){const p=await b.newPage({viewport:{width:w,height:h},reducedMotion:'reduce'});await p.goto('http://127.0.0.1:3017');await p.waitForTimeout(300);const e=p.locator('.hero .note');const y=await e.evaluate(e=>e.getBoundingClientRect().top+scrollY);for(const pos of [.15,.45,.75,.9]){await p.evaluate(({y,pos,h})=>{document.documentElement.style.scrollBehavior='auto';scrollTo(0,Math.max(0,y-h*pos))},{y,pos,h});await p.waitForTimeout(100);const prefix=`/tmp/contrast-scroll-${w}-${pos}`;await e.screenshot({path:prefix+'-note.png'});const color=await e.evaluate(e=>getComputedStyle(e).color);await e.evaluate(e=>e.style.setProperty('-webkit-text-fill-color','transparent','important'));await e.screenshot({path:prefix+'-background.png'});await e.evaluate(e=>e.style.removeProperty('-webkit-text-fill-color'));rows.push({w,h,prefix,color,pos})}await p.close()}fs.writeFileSync('/tmp/contrast-scroll-rows.json',JSON.stringify(rows));await b.close()})()

/* Reproducible exports from the frozen source, no live-site access required. */
const {chromium}=require('playwright');
const http=require('http'),fs=require('fs'),path=require('path');
const root=__dirname;
const types={'.html':'text/html','.png':'image/png','.svg':'image/svg+xml'};
const server=http.createServer((req,res)=>{
 const url=new URL(req.url,'http://localhost');
 const relative=url.pathname==='/'?'source/index.html':url.pathname.slice(1);
 const file=path.resolve(root,relative);
 if(!file.startsWith(root+path.sep)||!fs.existsSync(file)||!fs.statSync(file).isFile()){res.writeHead(404);return res.end()}
 res.setHeader('Content-Type',types[path.extname(file)]||'application/octet-stream');fs.createReadStream(file).pipe(res);
});
const wait=ms=>new Promise(r=>setTimeout(r,ms));
const props=['color','backgroundImage','fontFamily','fontSize','fontWeight','lineHeight','letterSpacing','padding','margin','borderRadius','maxWidth','transition','transform','width','height'];
const components={'navbar':'header','hero':'.hero','software-data-automation':'.eyebrow','selected-work':'.work','recorrido':'.process','me-pillas':'.play','sobre-mi':'.about','experiencia':'.experience','contacto':'.contact','footer':'footer'};
(async()=>{
 await new Promise(r=>server.listen(0,'127.0.0.1',r));
 const base='http://127.0.0.1:'+server.address().port;
 let browser;
 try{
 browser=await chromium.launch({headless:true});
 const manifest={createdAt:new Date().toISOString(),zoom:'100%',deviceScaleFactor:1,captures:[],unimplemented:['menu-mobile-open','sticky-navbar'],errors:[]};
 for(const [name,width,height] of [['desktop',1440,900],['tablet',768,1024],['mobile',390,844]]){
 const context=await browser.newContext({viewport:{width,height},deviceScaleFactor:1,reducedMotion:'no-preference'}),page=await context.newPage();
 page.on('pageerror',e=>manifest.errors.push({viewport:name,message:e.message}));
 await page.goto(base,{waitUntil:'load'});await page.evaluate(()=>document.fonts.ready);await wait(1100);
 const max=await page.evaluate(()=>document.documentElement.scrollHeight-innerHeight);
 for(let y=0;y<=max+height;y+=height*.65){await page.evaluate(y=>window.scrollTo({top:y,behavior:'instant'}),y);await wait(180)}
 await page.evaluate(()=>window.scrollTo({top:0,behavior:'instant'}));await wait(1100);
 const file='capturas/'+name+'-'+width+'-full-page.png';await page.screenshot({path:path.join(root,file),fullPage:true});manifest.captures.push(file);
 const computed=await page.evaluate(({components,props})=>Object.fromEntries(Object.entries(components).map(([key,selector])=>{const el=document.querySelector(selector),c=getComputedStyle(el),r=el.getBoundingClientRect();return [key,{rect:{x:r.x,y:r.y+scrollY,width:r.width,height:r.height},...Object.fromEntries(props.map(p=>[p,c[p]]))}]})),{components,props});
 fs.writeFileSync(path.join(root,'specs/computed-'+name+'.json'),JSON.stringify(computed,null,2));
 if(name!=='tablet'){
 for(const [key,sel] of Object.entries(components)){const loc=page.locator(sel);await loc.scrollIntoViewIfNeeded();await wait(1100);const f='capturas/'+name+'-'+key+'.png';await loc.screenshot({path:path.join(root,f)});manifest.captures.push(f)}
 for(let i=0;i<3;i++){await page.locator('[data-select="'+i+'"]').click();await wait(1100);const f='estados/'+name+'-project-'+i+'-active.png';await page.locator('.work').screenshot({path:path.join(root,f)});manifest.captures.push(f);await page.locator('[data-project="'+i+'"]').screenshot({path:path.join(root,'capturas/'+name+'-project-'+i+'.png')})}
 await page.locator('[data-select="0"]').click();await wait(1000);
 await page.locator('[data-project="0"]').hover();await page.locator('.work').screenshot({path:path.join(root,'estados/'+name+'-project-hover.png')});
 await page.locator('#contact-button').scrollIntoViewIfNeeded();await page.locator('#contact-button').hover();await page.locator('.contact').screenshot({path:path.join(root,'estados/'+name+'-button-hover.png')});
 for(let i=0;i<4;i++){await page.locator('.play').scrollIntoViewIfNeeded();await wait(750);await page.locator('.play').screenshot({path:path.join(root,'estados/'+name+'-me-pillas-'+i+'.png')});if(i<3)await page.locator('.catch').click()}
 await page.evaluate(()=>window.scrollTo({top:600,behavior:'instant'}));await wait(500);await page.screenshot({path:path.join(root,'estados/'+name+'-navbar-after-scroll.png')});
 }
 await context.close();
 }
 // One continuous recording demonstrates the real interactions at desktop size.
 const context=await browser.newContext({viewport:{width:1440,height:900},deviceScaleFactor:1,recordVideo:{dir:path.join(root,'motion'),size:{width:1440,height:900}}});
 const page=await context.newPage(),video=page.video();await page.goto(base,{waitUntil:'load'});await wait(1200);
 for(const [x,y] of [[1100,220],[1400,650],[760,750],[1050,350]]){await page.mouse.move(x,y,{steps:20});await wait(900)}
 const max=await page.evaluate(()=>document.documentElement.scrollHeight-innerHeight);
 for(let y=0;y<=max+120;y+=120){await page.evaluate(y=>window.scrollTo({top:y,behavior:'instant'}),y);await wait(120)}
 await page.locator('.work').scrollIntoViewIfNeeded();await wait(700);
 for(const i of [1,2,0]){await page.locator('[data-select="'+i+'"]').click();await wait(1300)}
 await page.locator('.play').scrollIntoViewIfNeeded();await wait(900);
 for(let i=0;i<3;i++){await page.locator('.catch').click();await wait(1400)}
 await context.close();await video.saveAs(path.join(root,'motion/recorrido-interacciones-desktop.webm'));const raw=await video.path();if(raw!==path.join(root,'motion/recorrido-interacciones-desktop.webm'))fs.unlinkSync(raw);
 fs.writeFileSync(path.join(root,'specs/capture-manifest.json'),JSON.stringify(manifest,null,2));
 fs.writeFileSync(path.join(root,'specs/export-status.json'),JSON.stringify({screenshots:'generated by capture.cjs',recording:'generated',unimplemented:manifest.unimplemented,errors:manifest.errors},null,2));
 console.log('Exports complete. Inspect screenshots and WebM before transferring.');
 }finally{if(browser)await browser.close();await new Promise(r=>server.close(r))}
})().catch(e=>{console.error(e);process.exitCode=1});
